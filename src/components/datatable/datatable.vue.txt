<script setup lang="ts">
import type { TableColumn } from "@nuxt/ui";
import { useApi } from "~/composable/useApi";

const props = defineProps<{
  endpoint: string;
  columns: TableColumn<any, any>[];
  limitOptions?: number[];
}>();

const emit = defineEmits<{
  (e: 'fetched', data: any[]): void;
}>();

const api = useApi();
const items = ref<any[]>([]);
const totalItems = ref(0);
const limitOptions = props.limitOptions ?? [10, 20, 50, 100];
const limit = ref(limitOptions[0]);
const start = ref(0);
const page = ref(1);

defineExpose({
  updateStart,
  updateLimit,
  getDataItem,
});

async function getDataItem() {
  try {
    const response = await api.get(props.endpoint, {
      params: {
        start: start.value,
        limit: limit.value,
      },
    });

    items.value = response.data.items || [];
    totalItems.value = response.data.total || 0;

    emit('fetched', items.value);
  } catch (error) {
    console.error("Error fetching items:", error);
  }
}

function updateStart(newStart: number) {
  start.value = newStart;
  getDataItem();
}

function updateLimit(newLimit: number) {
  limit.value = newLimit;
  start.value = 0;
  page.value = 1;
  getDataItem();
}

onMounted(() => {
  getDataItem();
});

// Add a sequence column in front of props.columns without overwriting the original
const sequenceColumn: TableColumn<any, any> = {
    accessorKey: 'sequence',
    header: '#',
    cell: ({ row }) => start.value + row.index + 1,
};

const displayColumns = computed(() => [sequenceColumn, ...props.columns]);

</script>

<template>
  <div>
    <div class="border border-default rounded">
      <UTable
        :data="items"
        :columns="displayColumns"
        :pagination-options="{ manualPagination: true }"
        :striped="true"
        :hoverable="true"
      />
    </div>
    <div class="mt-4 flex justify-between items-center">
      <USelect
        v-model="limit"
        :items="limitOptions"
        label="Items per page"
        @update:model-value="updateLimit($event)"
      />
      <UPagination
        v-if="totalItems > 0"
        :total="totalItems"
        :items-per-page="limit"
        :page="page"
        @update:page="(p) => { updateStart((p - 1) * (limit ?? 10)); page = p }"
      />
    </div>
  </div>
</template>
